cmake_minimum_required(VERSION 3.18)
project(SeismicX_Picker_C_Backend VERSION 0.2.1 LANGUAGES C)
option(BUILD_SHARED_LIBS "Build shared library; OFF makes a static edge library" ON)
option(PNSN_OPENMP "Enable optional OpenMP parallelism" ON)
option(PNSN_CUDNN "Build optional vendor cuDNN backend (CUDA SDK + cuDNN required)" OFF)
set(CUDNN_ROOT "" CACHE PATH "cuDNN install prefix containing include and lib")
option(PNSN_CUDA "Build optional custom CUDA backend" OFF)
option(PNSN_SANITIZE "Enable address/undefined sanitizers" OFF)
option(PNSN_TESTS "Build C tests" ON)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(pnsn src/pnsn.c src/cpu.c src/heap.c)
target_compile_definitions(pnsn PRIVATE PNSN_BUILD)
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(pnsn PUBLIC PNSN_STATIC)
endif()
target_include_directories(pnsn PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(NOT MSVC)
  target_compile_options(pnsn PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(pnsn PRIVATE m)
endif()
if(PNSN_OPENMP)
  find_package(OpenMP QUIET COMPONENTS C)
  if(OpenMP_C_FOUND)
    target_link_libraries(pnsn PRIVATE OpenMP::OpenMP_C)
  endif()
endif()
if(PNSN_SANITIZE AND NOT MSVC)
  target_compile_options(pnsn PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pnsn PRIVATE -fsanitize=address,undefined)
endif()
if(PNSN_CUDA OR PNSN_CUDNN)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
if(PNSN_CUDA)
  add_library(pnsn_cuda SHARED src/pnsn.c src/cpu.c src/heap.c src/cuda.cu)
  target_include_directories(pnsn_cuda PUBLIC include)
  target_compile_definitions(pnsn_cuda PRIVATE PNSN_BUILD PNSN_CUDA)
  target_link_libraries(pnsn_cuda PRIVATE CUDA::cudart_static)
  if(NOT MSVC)
    target_link_libraries(pnsn_cuda PRIVATE m)
  endif()
  set_target_properties(pnsn_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
if(PNSN_CUDNN)
  find_path(CUDNN_INCLUDE_DIR cudnn.h HINTS "${CUDNN_ROOT}/include" "${CUDAToolkit_INCLUDE_DIRS}")
  find_library(CUDNN_LIBRARY NAMES cudnn libcudnn.so.9 cudnn64_9 HINTS "${CUDNN_ROOT}/lib" "${CUDNN_ROOT}/lib64")
  if(NOT CUDNN_INCLUDE_DIR OR NOT CUDNN_LIBRARY)
    message(FATAL_ERROR "cuDNN not found: set CUDNN_ROOT or CUDNN_INCLUDE_DIR and CUDNN_LIBRARY")
  endif()
  add_library(pnsn_cudnn SHARED src/pnsn.c src/cpu.c src/heap.c src/cudnn.cu)
  target_include_directories(pnsn_cudnn PUBLIC include PRIVATE "${CUDNN_INCLUDE_DIR}")
  target_compile_definitions(pnsn_cudnn PRIVATE PNSN_BUILD PNSN_CUDA PNSN_CUDNN)
  target_link_libraries(pnsn_cudnn PRIVATE CUDA::cudart_static "${CUDNN_LIBRARY}")
  if(NOT MSVC)
    target_link_libraries(pnsn_cudnn PRIVATE m)
  endif()
  set_target_properties(pnsn_cudnn PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_library(pnsn_cudnn_int8 SHARED src/pnsn.c src/cpu.c src/heap.c src/cudnn.cu)
  target_include_directories(pnsn_cudnn_int8 PUBLIC include PRIVATE "${CUDNN_INCLUDE_DIR}")
  target_compile_definitions(pnsn_cudnn_int8 PRIVATE PNSN_BUILD PNSN_CUDA PNSN_CUDNN PNSN_CUDNN_INT8)
  target_link_libraries(pnsn_cudnn_int8 PRIVATE CUDA::cudart_static "${CUDNN_LIBRARY}")
  if(NOT MSVC)
    target_link_libraries(pnsn_cudnn_int8 PRIVATE m)
  endif()
  set_target_properties(pnsn_cudnn_int8 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_executable(pnsn_cudnn_int8_cli src/cli.c)
  target_link_libraries(pnsn_cudnn_int8_cli PRIVATE pnsn_cudnn_int8)
  install(TARGETS pnsn_cudnn_int8 pnsn_cudnn_int8_cli LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  add_library(SeismicX::CuDNN ALIAS pnsn_cudnn)
  add_library(SeismicX::CuDNN_INT8 ALIAS pnsn_cudnn_int8)
  add_executable(pnsn_cudnn_cli src/cli.c)
  target_link_libraries(pnsn_cudnn_cli PRIVATE pnsn_cudnn)
  install(TARGETS pnsn_cudnn pnsn_cudnn_cli LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_library(SeismicX::CPU ALIAS pnsn)
add_executable(seismicx_cli src/cli.c)
target_link_libraries(seismicx_cli PRIVATE pnsn)
add_executable(pnsn_cli src/cli.c)
target_link_libraries(pnsn_cli PRIVATE pnsn)
if(PNSN_SANITIZE AND NOT MSVC)
  target_compile_options(pnsn_cli PRIVATE -fsanitize=address,undefined)
  target_link_options(pnsn_cli PRIVATE -fsanitize=address,undefined)
  target_compile_options(seismicx_cli PRIVATE -fsanitize=address,undefined)
  target_link_options(seismicx_cli PRIVATE -fsanitize=address,undefined)
endif()
if(PNSN_CUDA)
  add_executable(pnsn_cuda_cli src/cli.c)
  target_link_libraries(pnsn_cuda_cli PRIVATE pnsn_cuda)
endif()
if(PNSN_TESTS)
  enable_testing()
  add_executable(pnsn_unit tests/unit.c)
  target_link_libraries(pnsn_unit PRIVATE pnsn)
  if(PNSN_SANITIZE AND NOT MSVC)
    target_compile_options(pnsn_unit PRIVATE -fsanitize=address,undefined)
    target_link_options(pnsn_unit PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME pnsn_unit COMMAND pnsn_unit)
  add_executable(pnsn_batch_unit tests/batch_unit.c)
  target_link_libraries(pnsn_batch_unit PRIVATE pnsn)
  if(PNSN_SANITIZE AND NOT MSVC)
    target_compile_options(pnsn_batch_unit PRIVATE -fsanitize=address,undefined)
    target_link_options(pnsn_batch_unit PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME pnsn_batch_unit COMMAND pnsn_batch_unit "${CMAKE_CURRENT_SOURCE_DIR}/models/pnsn-int8.bin")
  set_tests_properties(pnsn_batch_unit PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME seismicx_unet_batch_unit COMMAND pnsn_batch_unit "${CMAKE_CURRENT_SOURCE_DIR}/models/china-unet-int8.bin")
  set_tests_properties(seismicx_unet_batch_unit PROPERTIES SKIP_RETURN_CODE 77)
  if(PNSN_CUDNN)
    add_executable(pnsn_cudnn_int8_ops tests/cudnn_int8_ops.cu src/pnsn.c src/cpu.c src/heap.c)
    target_include_directories(pnsn_cudnn_int8_ops PRIVATE include "${CUDNN_INCLUDE_DIR}")
    target_compile_definitions(pnsn_cudnn_int8_ops PRIVATE PNSN_BUILD PNSN_CUDA PNSN_CUDNN PNSN_CUDNN_INT8)
    target_link_libraries(pnsn_cudnn_int8_ops PRIVATE CUDA::cudart_static "${CUDNN_LIBRARY}")
    if(NOT MSVC)
      target_link_libraries(pnsn_cudnn_int8_ops PRIVATE m)
    endif()
    set_target_properties(pnsn_cudnn_int8_ops PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    add_test(NAME pnsn_cudnn_int8_ops COMMAND pnsn_cudnn_int8_ops "${CMAKE_CURRENT_SOURCE_DIR}/models/pnsn-int8.bin")
    set_tests_properties(pnsn_cudnn_int8_ops PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME seismicx_unet_int8_ops COMMAND pnsn_cudnn_int8_ops "${CMAKE_CURRENT_SOURCE_DIR}/models/china-unet-int8.bin")
    set_tests_properties(seismicx_unet_int8_ops PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
install(TARGETS pnsn pnsn_cli seismicx_cli LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/pnsn.h include/seismicx.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
